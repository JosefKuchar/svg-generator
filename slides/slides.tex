\section{Motivation}

\begin{frame}{Why Generate SVG?}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.52\textwidth}
    \begin{itemize}
      \tightitems
      \item SVG is the native format for icons, logos, illustrations, diagrams, and UI assets.
      \item Vector graphics are resolution independent, compact, and editable.
      \item Modern text-to-image models are strong, but their output is still a bitmap.
      \item The target artifact is not only an image; it is a structured object that can be modified.
    \end{itemize}
  \end{column}
  \begin{column}{.42\textwidth}
    \centering
    \includegraphics[width=.95\linewidth,height=.62\textheight,keepaspectratio]{svg_primitives_example.pdf}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Central Problem}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.5\textwidth}
    Direct text-to-SVG generation must solve several hard problems at once:
    \begin{itemize}
      \tightitems
      \item semantic grounding from text,
      \item visual composition,
      \item geometric structure,
      \item valid SVG syntax and rendering behavior.
    \end{itemize}
  \end{column}
  \begin{column}{.44\textwidth}
    \begin{block}{Vectorization is ambiguous}
      Many different sets of paths, fills, and curves can render to nearly the same pixels.
    \end{block}
    \vspace{.6em}
    A useful model must balance raster fidelity with validity, compactness, and editability.
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Objective}
\begin{block}{Goal}
Design, implement, and evaluate a modular text-to-SVG pipeline.
\end{block}

\vspace{.6em}
\begin{itemize}
  \tightitems
  \item Adapt a pretrained raster generator to produce SVG-like images.
  \item Train a custom raster-to-vector model with synthetic and SVG-derived supervision.
  \item Compare flow matching with autoregressive and classical vectorization baselines.
  \item Evaluate not only rendered similarity, but also SVG validity, compactness, and editability.
\end{itemize}
\end{frame}

\section{Vectorization}

\begin{frame}{Stage 2: Bezier Representation}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.5\textwidth}
    \begin{itemize}
      \tightitems
      \item The model predicts a fixed-size representation of cubic Bezier curves.
      \item Fixed dimensionality makes flow matching practical.
      \item The representation directly renders to SVG paths.
      \item Capacity and primitive semantics are intentionally constrained.
    \end{itemize}
  \end{column}

\begin{frame}{What the Results Show}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.31\textwidth}
    \begin{block}{vtracer}
      Best raster reconstruction, but large SVGs with many paths and commands.
    \end{block}
  \end{column}
  \begin{column}{.31\textwidth}
    \begin{block}{Autoregressive baselines}
      Sometimes compact and faithful, but less reliable and more prone to invalid outputs.
    \end{block}
  \end{column}
  \begin{column}{.31\textwidth}
    \begin{block}{Flow vectorizer}
      Lower pixel fidelity than tracing, but compact and valid across tested settings.
    \end{block}
  \end{column}
\end{columns}

\vspace{.8em}
\centering
\structure{The contribution is not a better tracer; it is a step toward neural SVGs as structured editable objects.}
\end{frame}

\section{Conclusion}

\begin{frame}{Contributions}
\begin{itemize}
  \tightitems
  \item Implemented a two-stage text-to-SVG pipeline using adapted raster generation and raster-to-vector conversion.
  \item Designed a conditional flow-matching vectorizer over a fixed Bezier representation.
  \item Built a synthetic Bezier data generator for supervised vectorizer pretraining.
  \item Evaluated SVG generation with fidelity, validity, compactness, and editability-oriented metrics.
\end{itemize}
\end{frame}

\begin{frame}{Limitations and Future Work}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.48\textwidth}
    \structure{Current limitations}
    \begin{itemize}
      \tightitems
      \item Bezier-only geometry loses primitive semantics.
      \item No gradients, masks, filters, or complex style rules.
      \item Fixed segment budget limits complex graphics.
      \item Generated rasters remain a source of artifacts.
    \end{itemize}
  \end{column}
  \begin{column}{.48\textwidth}
    \structure{Next steps}
    \begin{itemize}
      \tightitems
      \item Recover higher-level primitives after Bezier prediction.
      \item Extend the representation to richer SVG styling.
      \item Train on larger paired vector datasets.
      \item Improve transparent and foreground-asset generation.
    \end{itemize}
  \end{column}
\end{columns}
\end{frame}
